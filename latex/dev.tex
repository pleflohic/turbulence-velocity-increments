\section{Starting point: the distribution of $h$}

We start from Equation~(11) of~[1], which defines the probability density function of the local Hölder exponent \(h\):
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\left(\tfrac{\ell}{L}\right)^{1-\mathcal{D}^E(h)}
}{
\displaystyle \int_{h_{\min}}^{h_{\max}}
\left(\tfrac{\ell}{L}\right)^{1-\mathcal{D}^E(h)}\,\mathrm{d}h
}.
\tag{1}
\end{equation}

Using the Kolmogorov–Oboukhov log-normal spectrum
\[
\mathcal{D}^E(h)=1-\frac{(h-c_1)^2}{2c_2},
\]
with \(h_{\min}=-\infty\) and \(h_{\max}=+\infty\),
we obtain
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\exp\!\left[-\tfrac{(h - c_1)^2}{2c_2}\ln\!\left(\tfrac{L}{\ell}\right)\right]
}{
\displaystyle \int_{-\infty}^{+\infty}
\exp\!\left[-\tfrac{(h - c_1)^2}{2c_2}\ln\!\left(\tfrac{L}{\ell}\right)\right]
\mathrm{d}h
}.
\tag{2}
\end{equation}

To simplify notation, we define
\begin{equation}
\alpha = -\ln\!\left(\frac{\ell}{L}\right) > 0.
\tag{3}
\end{equation}
Then the expression becomes
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right]
}{
\displaystyle \int_{-\infty}^{+\infty}
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right]
\mathrm{d}h
}.
\tag{4}
\end{equation}

The denominator is a Gaussian integral:
\begin{equation}
\int_{-\infty}^{+\infty}
e^{-\frac{\alpha}{2c_2}(h - c_1)^2}\mathrm{d}h
= \sqrt{\frac{2\pi c_2}{\alpha}}.
\tag{5}
\end{equation}
Hence,
\begin{equation}
\boxed{
\mathcal{P}_h^{(\ell)}(h)
=
\sqrt{\frac{\alpha}{2\pi c_2}}\,
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right].
}
\tag{6}
\end{equation}

\section{Computation of the PDF using Laplace's method}

We now consider the probability density function of velocity increments:
\begin{equation}
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
= \frac{1}{2\pi\sigma}\,
\sqrt{\frac{\alpha}{c_2}}
\int_{-\infty}^{+\infty}
\exp[-f(h,\alpha,\delta_\ell u)]\,\mathrm{d}h.
\tag{7}
\end{equation}
with
\begin{equation}
f(h,\alpha,\delta_\ell u)
= \alpha\!\left(\frac{(h-c_1)^2}{2c_2}-h\right)
+ \frac{1}{2}\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}.
\tag{8}
\end{equation}

\paragraph{Laplace approximation and signs.}
The integral is dominated by the minimum of \(f(h,\alpha,\delta_\ell u)\).
A quadratic expansion around the stationary point \(h^\star\) gives
\begin{equation}
\int_{-\infty}^{+\infty}\exp[-f(h,\alpha,\delta_\ell u)]\,\mathrm{d}h
\approx
\exp[-f(h^\star,\alpha,\delta_\ell u)]
\sqrt{\frac{2\pi}{f''(h^\star,\alpha,\delta_\ell u)}}.
\tag{9}
\end{equation}
Therefore
\begin{equation}
\boxed{
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
\approx
\frac{1}{2\pi\sigma}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{\frac{2\pi}{f''(h^\star,\alpha,\delta_\ell u)}}\,
\exp[-f(h^\star,\alpha,\delta_\ell u)].
}
\tag{10}
\end{equation}

Since \(\alpha>0\), the term \(\alpha (h-c_1)^2/(2c_2)\) grows quadratically for large \(|h|\),
and the exponential term is positive.
Thus \(f(h,\alpha,\delta_\ell u)\to+\infty\) as \(|h|\to\infty\),
so the stationary point is a \emph{minimum} and \(f''(h^\star,\alpha,\delta_\ell u)>0\),
which justifies the steepest descent method.

\subsection{Case A: assuming \(h^\star = c_1\)}

From \((8)\),
\begin{equation}
f(h^\star,\alpha,\delta_\ell u)\Big|_{h^\star=c_1}
= \alpha c_1
+ \frac{1}{2}\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1},
\tag{11}
\end{equation}
and
\begin{equation}
f''(h^\star,\alpha,\delta_\ell u)\Big|_{h^\star=c_1}
= \frac{\alpha}{c_2}
+2\alpha^2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1}.
\tag{12}
\end{equation}
Plug into \((10)\):
\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
&\approx
\frac{1}{2\pi\sigma}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{
\frac{2\pi}{
\frac{\alpha}{c_2}
+2\alpha^2
\left(\tfrac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1}
}
}
\\[0.8em]
&\quad\times
\exp\!\left[
-\alpha c_1
-\frac{1}{2}\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1}
\right].
\end{aligned}
}
\tag{13}
\end{equation}

\subsection{Case B: solving \(f'(h,\alpha,\delta_\ell u)=0\) via the Lambert \(W\) function}

From \((8)\),
\begin{equation}
\frac{\partial f}{\partial h}(h,\alpha,\delta_\ell u)
= \alpha\!\left(\frac{h-c_1}{c_2}-1\right)
+ \alpha\!\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}.
\tag{14}
\end{equation}
Set to zero:
\begin{equation}
h - (c_1 + c_2)
= -c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}.
\tag{15}
\end{equation}
Multiply by \(-2\alpha e^{-2\alpha h}\):
\begin{equation}
(-2\alpha)(h - c_1 - c_2)e^{-2\alpha h}
= 2\alpha c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}.
\tag{16}
\end{equation}
Let
\begin{equation}
u = -2\alpha(h - c_1 - c_2)
\quad\Longrightarrow\quad
u e^{u}
= 2\alpha c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}.
\tag{17}
\end{equation}
Hence
\begin{equation}
\boxed{
h^\star
= (c_1 + c_2)
- \frac{1}{2\alpha}
W\!\left(
2\alpha c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}
\right),
}
\tag{18}
\end{equation}
where \(W(x)\) is the Lambert \(W\) function (the inverse of \(w\mapsto we^w\)).

Moreover,
\begin{equation}
f''(h,\alpha,\delta_\ell u)
= \frac{\alpha}{c_2}
+ 2\alpha^2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}.
\tag{19}
\end{equation}
Insert \(h=h^\star\) into \((10)\) to obtain:
\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
&\approx
\frac{1}{2\pi\sigma}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{
\frac{2\pi}{
\frac{\alpha}{c_2}
+2\alpha^2
\left(\tfrac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h^\star}
}
}
\\[0.8em]
&\quad\times
\exp\!\left[
-\alpha\!\left(
\frac{(h^\star-c_1)^2}{2c_2}-h^\star
\right)
-\frac{1}{2}
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h^\star}
\right].
\end{aligned}
}
\tag{20}
\end{equation}

\subsection{Normalized increment by the order 2 moment}

From the multifractal formalism, the $p$-th order moment of the velocity increment
is expressed as
\begin{equation}
\langle |\delta_\ell u|^p \rangle \sim \ell^{\zeta_p},
\quad\text{with}\quad
\zeta_p = \min_h \bigl[ p h + 1 - \mathcal{D}^E(h) \bigr].
\tag{21}
\end{equation}

Using the Kolmogorov–Oboukhov log-normal spectrum
\begin{equation}
\mathcal{D}^E(h) = 1 - \frac{(h - c_1)^2}{2c_2},
\tag{22}
\end{equation}
the minimization of Eq.~(21) gives
\begin{equation}
\frac{\mathrm{d}}{\mathrm{d}h}
\bigl[p h + 1 - \mathcal{D}^E(h)\bigr] = 0
\quad\Longrightarrow\quad
p = \frac{h - c_1}{c_2}.
\tag{23}
\end{equation}

Hence, the minimizing exponent is
\begin{equation}
h_p = c_1 + p c_2,
\tag{24}
\end{equation}
and inserting this value into Eq.~(21) gives
\begin{equation}
\zeta_p = p c_1 + c_2 p^2.
\tag{25}
\end{equation}

In particular, for the second-order moment ($p=2$):
\begin{equation}
\boxed{
\langle (\delta_\ell u)^2 \rangle
\sim
\left( \frac{\ell}{L} \right)^{2c_1 + 4c_2}.
}
\tag{26}
\end{equation}

We now define a normalized increment
\begin{equation}
\boxed{
\delta_\ell \nu
= \frac{\delta_\ell u}{\sqrt{\langle (\delta_\ell u)^2 \rangle}}
= \delta_\ell u
\left( \frac{L}{\ell} \right)^{c_1 + 2c_2}.
}
\tag{27}
\end{equation}

Replacing $\delta_\ell u = \delta_\ell \nu\, (\ell/L)^{c_1 + 2c_2}$
in Eq.~(20) gives the normalized form of the PDF:
\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell \nu}(\delta_\ell \nu)
&\approx
\frac{1}{2\pi}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{
\frac{2\pi}{
\frac{\alpha}{c_2}
+2\alpha^2
(\delta_\ell \nu)^{2}
e^{2\alpha h^\star}
}
}
\\[0.8em]
&\quad\times
\exp\!\left[
-\alpha\!\left(
\frac{(h^\star-c_1)^2}{2c_2}-h^\star
\right)
-\frac{1}{2}
(\delta_\ell \nu)^{2}
e^{2\alpha h^\star}
\right],
\end{aligned}
}
\tag{28}
\end{equation}
where $h^\star$ is still determined by Eq.~(18)
with $\delta_\ell u$ replaced by $\delta_\ell \nu\, (\ell/L)^{c_1 + 2c_2}$.

This normalization ensures that
\begin{equation}
\langle (\delta_\ell \nu)^2 \rangle = 1,
\tag{29}
\end{equation}
for all scales $\ell/L$, so that all normalized PDFs can be directly compared across scales.
