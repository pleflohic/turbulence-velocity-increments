\section{Starting point: the distribution of $h$}

We start from Equation~(11) of~[1], which defines the probability density function of the local Hölder exponent \(h\):
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\left(\tfrac{\ell}{L}\right)^{1-\mathcal{D}^E(h)}
}{
\displaystyle \int_{h_{\min}}^{h_{\max}}
\left(\tfrac{\ell}{L}\right)^{1-\mathcal{D}^E(h)}\,\mathrm{d}h
}.
\tag{1}
\end{equation}

Using the Kolmogorov–Oboukhov log-normal spectrum
\[
\mathcal{D}^E(h)=1-\frac{(h-c_1)^2}{2c_2},
\]
with \(h_{\min}=-\infty\) and \(h_{\max}=+\infty\),
we obtain
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\exp\!\left[-\tfrac{(h - c_1)^2}{2c_2}\ln\!\left(\tfrac{L}{\ell}\right)\right]
}{
\displaystyle \int_{-\infty}^{+\infty}
\exp\!\left[-\tfrac{(h - c_1)^2}{2c_2}\ln\!\left(\tfrac{L}{\ell}\right)\right]
\mathrm{d}h
}.
\tag{2}
\end{equation}

To simplify notation, we define
\begin{equation}
\alpha = -\ln\!\left(\frac{\ell}{L}\right) > 0.
\tag{3}
\end{equation}
Then the expression becomes
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right]
}{
\displaystyle \int_{-\infty}^{+\infty}
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right]
\mathrm{d}h
}.
\tag{4}
\end{equation}

The denominator is a Gaussian integral:
\begin{equation}
\int_{-\infty}^{+\infty}
e^{-\frac{\alpha}{2c_2}(h - c_1)^2}\mathrm{d}h
= \sqrt{\frac{2\pi c_2}{\alpha}}.
\tag{5}
\end{equation}
Hence,
\begin{equation}
\boxed{
\mathcal{P}_h^{(\ell)}(h)
=
\sqrt{\frac{\alpha}{2\pi c_2}}\,
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right].
}
\tag{6}
\end{equation}

\section{Computation of the PDF using Laplace's method}

\subsection{General form of the PDF}

We now consider the probability density function of velocity increments:
\begin{equation}
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
= \frac{1}{2\pi\sigma}\,
\sqrt{\frac{\alpha}{c_2}}
\int_{-\infty}^{+\infty}
\exp[-f(h,\alpha,\delta_\ell u)]\,\mathrm{d}h.
\tag{7}
\end{equation}
with
\begin{equation}
f(h,\alpha,\delta_\ell u)
= \alpha\!\left(\frac{(h-c_1)^2}{2c_2}-h\right)
+ \frac{1}{2}\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{8}
\end{equation}

\paragraph{Laplace approximation}
Let $I=[a,b]$ be an interval, $\varphi\in C^2(I,\mathbb{R})$
and $g\in C^0(I,\mathbb{C})$. Assume that:
\begin{enumerate}
    \item[(i)] For all $t>0$, $\displaystyle \int_a^b e^{t\varphi(x)} |g(x)|\,dx < +\infty$;
    \item[(ii)] $\varphi'$ vanishes at a unique point $x_0\in I$ and $\varphi''(x_0)<0$;
    \item[(iii)] $g(x_0)\neq 0$.
\end{enumerate}
Then
\[
\int_a^b e^{\,t\varphi(x)}\, g(x)\,dx
\sim
\sqrt{\frac{2\pi}{t\,|\varphi''(x_0)|}}\,
e^{\,t\varphi(x_0)}\, g(x_0)
\qquad\text{as } t\to +\infty .\]



$\\[0.8em]$
Knowing that
\begin{equation}
\forall h \in \mathbf{R},
\quad
\frac{\partial^2 f}{\partial h^2}(h,\alpha,\delta_\ell u)
=
\frac{\alpha}{c_2}
+ 2\alpha^2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}
\quad > 0
\tag{11}
\end{equation}

we just need to verify that $h^\star$ is such that
$\frac{\partial f}{\partial h}(h^\star,\alpha,\delta_\ell u) = 0$
to be able to use Laplace approximation.

$\\[0.3em]$
In this case, by applying the approximation to Eq (7), we get :

\begin{equation}
\boxed{
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
\approx
\frac{1}{2\pi\sigma}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{\frac{2\pi}{\frac{\partial^2 f}{\partial h^2}(h,\alpha,\delta_\ell u))}}\,
\exp[-f(h^\star,\alpha,\delta_\ell u)].
}
\tag{10}
\end{equation}


$\\[0.1em]$
i.e
\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
\approx
\frac{1}{2\pi\sigma}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{
\frac{2\pi}{
\frac{\alpha}{c_2}
+2\alpha^2
e^{2\alpha h^\star}
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
}
}
\times
\exp\!\left[
-\alpha\!\left(
\frac{(h^\star-c_1)^2}{2c_2}-h^\star
\right)
-\frac{1}{2}
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h^\star}
\right]
\end{aligned}
}
\tag{12}
\end{equation}


\subsection{Case A: assuming \(h^\star = c_1\)}

As a first approximation, we can consider that :

\begin{equation}
\frac{\partial f}{\partial h}(h^\star,\alpha,\delta_\ell u)\Big|_{h^\star=c_1}
= 0
\tag{13}
\end{equation}

$\\[0.1em]$
And since

\begin{equation}
\frac{\partial^2 f}{\partial h^2}(h^\star,\alpha,\delta_\ell u)\Big|_{h^\star=c_1}
= \frac{\alpha}{c_2}
+2\alpha^2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1}
\quad > 0
\tag{14}
\end{equation}

$\\[0.3em]$
We can use the Laplace approximation. Eq(12) then becomes :

\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell u}(\delta_\ell u)
\approx
\frac{1}{2\pi\sigma}
\sqrt{\frac{\alpha}{c_2}}\,
\sqrt{
\frac{2\pi}{
\frac{\alpha}{c_2}
+2\alpha^2
\left(\tfrac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1}
}
}
\times
\exp\!\left[
\alpha c_1
-\frac{1}{2}\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha c_1}
\right]
\end{aligned}
}
\tag{15}
\end{equation}

\subsection{Case B: solving \(\frac{\partial f}{\partial h}(h,\alpha,\delta_\ell u)=0\) via the Lambert \(W\) function}

From \((8)\),
\begin{equation}
\frac{\partial f}{\partial h}(h,\alpha,\delta_\ell u)
= \alpha\!\left(\frac{h-c_1}{c_2}-1\right)
+ \alpha\!\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{16}
\end{equation}
Set to zero:
\begin{equation}
h - (c_1 + c_2)
= -c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{17}
\end{equation}
Multiply by \(-2\alpha\) and let \(u=-2\alpha(h-c_1-c_2)\):
\begin{equation}
u
= 2\alpha c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)-u}
\tag{18}
\end{equation}
Consequently
\begin{equation}
u e^{u}
= 2\alpha c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}
\tag{19}
\end{equation}
Hence
\begin{equation}
\boxed{
h^\star
= (c_1 + c_2)
- \frac{1}{2\alpha}
W\!\left(
2\alpha c_2
\left(\frac{\delta_\ell u}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}
\right)
}
\tag{20}
\end{equation}
where \(W(x)\) is the Lambert \(W\) function (the inverse of \(w\mapsto we^w\)).

$\\[0.1em]$
We can now plug the obtained $h^\star$ into Eq(12).


\section{Normalized PDF}

The $p$-th order moment of the velocity increment
is expressed as
\begin{equation}
\langle |\delta_\ell u|^p \rangle \sim \ell^{\zeta_p},
\quad\text{with}\quad
\zeta_p = \min_h \bigl[ p h + 1 - \mathcal{D}^E(h) \bigr].
\tag{21}
\end{equation}

Using the Kolmogorov–Oboukhov log-normal spectrum
\begin{equation}
\mathcal{D}^E(h) = 1 - \frac{(h - c_1)^2}{2c_2},
\tag{14}
\end{equation}
the minimization of Eq.~(21) gives
\begin{equation}
\frac{\mathrm{d}}{\mathrm{d}h}
\bigl[p h + 1 - \mathcal{D}^E(h)\bigr] = 0
\quad i.e. \quad
\frac{\mathrm{d}}{\mathrm{d}h}
\bigl[p h + \frac{(h-c_1)^2}{2c_2}\bigr] = 0
\tag{22}
\end{equation}

Hence, the minimizing exponent is
\begin{equation}
h_p = c_1 - p c_2,
\tag{23}
\end{equation}
and inserting this value into Eq.~(21) gives
\begin{equation}
\zeta_p = p c_1 - \frac{p^2}{2} c_2
\tag{24}
\end{equation}

In particular, for the second-order moment ($p=2$):
\begin{equation}
\boxed{
\langle (\delta_\ell u)^2 \rangle
\sim
\ell^{2(c_1 - c_2)}.
}
\tag{25}
\end{equation}

We now define a normalized increment
\begin{equation}
\boxed{
\delta_\ell \nu
= \frac{\delta_\ell u}{\sqrt{\langle (\delta_\ell u)^2 \rangle}}
= \frac{\delta_\ell u}{\ell^{c_1 - c_2}}
}
\tag{26}
\end{equation}

$\\[0.3em]$
This normalization ensures that
\begin{equation}
\langle (\delta_\ell \nu)^2 \rangle = 1,
\tag{27}
\end{equation}
for all scales $\ell/L$, so that all normalized PDFs can be directly compared across scales.

$\\[0.1em]$
Replacing $\delta_\ell u = \delta_\ell \nu\, \ell^{c_1 - c_2}$
in Eq. (12) gives the normalized form of the PDF :
\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell \nu}(\delta_\ell \nu)
\approx
\frac{1}{\sqrt{2\pi}}
\frac{1}{
\sqrt{
2c_2 \alpha \ell^{2(c_1 - c_2)} e^{2\alpha h^\star} {\delta_\ell \nu}^2 + \sigma^2
}
}
\times
\exp\!\left[
-\alpha\!\left(
\frac{(h^\star-c_1)^2}{2c_2}-h^\star
\right)
- \frac{1}{2}
\left(\frac{
\delta_\ell \nu
}{\sigma}\right)^{2}
\ell^{2(c_1 - c_2)} e^{2\alpha h^\star}
\right]
\end{aligned}
}
\tag{28}
\end{equation}



\subsection{Case A: assuming \(h^\star = c_1\)}

As a first approximation, we can consider that :

\begin{equation}
\frac{\partial f}{\partial h}(h^\star,\alpha,\delta_\ell \nu)\Big|_{h^\star=c_1}
= 0
\tag{29}
\end{equation}

And since

\begin{equation}
\frac{\partial^2 f}{\partial h^2}(h^\star,\alpha,\delta_\ell \nu)\Big|_{h^\star=c_1}
= \frac{\alpha}{c_2}
+2\alpha^2
\left(\frac{\delta_\ell \nu\,\ell^{c_1 - c_2}}{\sigma}\right)^{2}
e^{2\alpha c_1}
\quad > 0
\tag{30}
\end{equation}

We can use the Laplace approximation. Eq(28) then becomes :

\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell \nu}(\delta_\ell \nu)
\approx
\frac{1}{\sqrt{2\pi}}
\frac{1}{
\sqrt{
2c_2\alpha\,
\left(\frac{L^{c_1}}{\ell^{c_2}}\right)^{2}
(\delta_\ell \nu)^2
+\sigma^2
}
}
\times
\exp\!\left[
\alpha c_1
-\frac{1}{2}
\left(\frac{L^{c_1}}{\ell^{c_2}}\right)^{2}
\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\right]
\end{aligned}
}
\tag{31}
\end{equation}



\subsection{Case B: solving \(\frac{\partial f}{\partial h}(h,\alpha,\delta_\ell \nu)=0\) via the Lambert \(W\) function}

From \((8)\),
\begin{equation}
\frac{\partial f}{\partial h}(h,\alpha,\delta_\ell \nu)
= \alpha\!\left(\frac{h-c_1}{c_2}-1\right)
+ \alpha\!\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\ell^{2(c_1 - c_2)}
e^{2\alpha h}
\tag{32}
\end{equation}

Set to zero:
\begin{equation}
h - (c_1 + c_2)
= -c_2
\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\ell^{2(c_1 - c_2)}
e^{2\alpha h}
\tag{33}
\end{equation}

Multiply by \(-2\alpha\) and let \(u=-2\alpha(h-c_1-c_2)\):
\begin{equation}
u
= 2\alpha c_2
\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\ell^{2(c_1 - c_2)}
e^{2\alpha(c_1 + c_2)-u}
\tag{34}
\end{equation}

Consequently
\begin{equation}
u e^{u}
= 2\alpha c_2
\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\ell^{2(c_1 - c_2)}
e^{2\alpha(c_1 + c_2)}
\tag{35}
\end{equation}

Hence
\begin{equation}
h^\star
= (c_1 + c_2)
- \frac{1}{2\alpha}
W\!\left(
2\alpha c_2
\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\ell^{2(c_1 - c_2)}
e^{2\alpha(c_1 + c_2)}
\right)
\tag{36}
\end{equation}

i.e.
\begin{equation}
\boxed{
h^\star
= (c_1 + c_2)
- \frac{1}{2\alpha}
W\!\left(
2\alpha c_2
\left(\frac{L^{c_1 + c_2}}{\ell^{2c_2}}\right)^{2}
\left(\frac{\delta_\ell \nu}{\sigma}\right)^{2}
\right)
}
\tag{37}
\end{equation}

We can now plug the obtained $h^\star$ into Eq(28).




\section{Entropy of the normalized PDF}

We now compute the Shannon entropy of the normalized increment PDF
\(\mathcal{P}_{\delta_\ell \nu}\) obtained in Eq. (28). By definition,
\begin{equation}
S(\delta_\ell \nu)
=
- \int_{-\infty}^{+\infty}
\mathcal{P}_{\delta_\ell \nu}(x)\,
\ln\!\bigl[\mathcal{P}_{\delta_\ell \nu}(x)\bigr]\,
\mathrm{d}x
\tag{36}
\end{equation}
