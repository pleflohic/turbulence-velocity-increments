\title{Evaluation of increment PDFs using Laplace's method}
\author{}
\date{}
\maketitle

\begin{abstract}
\normalsize

We study the probability density functions (PDFs) of Eulerian velocity increments in fully developed turbulence. The work is based on the multifractal propagator formalism and follows the phenomenological framework of Chevillard \emph{et al.} We focus on the Kolmogorov--Oboukhov log-normal model. In this picture, large-scale velocity fluctuations are Gaussian, while intermittency enters through a distribution of local Hölder exponents. These ingredients are coupled by a scale-dependent propagator, from which explicit expressions for the increment PDFs are derived as integrals over the Hölder exponent.

The PDFs of the increments are written as integrals over the Hölder exponent, which cannot be evaluated in closed form. We therefore study their asymptotic behavior in the inertial-range limit of small scale separation using Laplace’s method. Two saddle-point approximations are examined. In the first one, the dominant Hölder exponent is assumed to be fixed. In the second one, the saddle point is obtained exactly by solving the stationarity condition with the Lambert \(W\) function. Both approximations are compared with direct numerical evaluations of the full propagator integral.

We also introduce a scale-dependent normalization based on the second-order structure function. This normalization allows a direct comparison of PDFs at different scales and separates variance scaling from genuine intermittency effects. Numerical results obtained from Julia notebooks show the evolution of the PDF shapes with scale, including logarithmic representations. At small scales, the Laplace approximation captures the dominant behavior of the full propagator. The remaining discrepancies quantify the limits of the reduced assumptions and clarify the range of validity of asymptotic treatments in the propagator framework, providing a controlled basis for subsequent entropy-based analyses of intermittency.
\end{abstract}

\section{Starting point: the distribution of $h$}

We start from Equation~(11) of~[1], which defines the probability density function of the local Hölder exponent \(h\):
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\left(\tfrac{\ell}{L}\right)^{1-\mathcal{D}^E(h)}
}{
\displaystyle \int_{h_{\min}}^{h_{\max}}
\left(\tfrac{\ell}{L}\right)^{1-\mathcal{D}^E(h)}\,\mathrm{d}h
}.
\tag{1}
\end{equation}

Using the Kolmogorov–Oboukhov log-normal spectrum
\[
\mathcal{D}^E(h)=1-\frac{(h-c_1)^2}{2c_2},
\]
with \(h_{\min}=-\infty\) and \(h_{\max}=+\infty\),
we obtain
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\exp\!\left[-\tfrac{(h - c_1)^2}{2c_2}\ln\!\left(\tfrac{L}{\ell}\right)\right]
}{
\displaystyle \int_{-\infty}^{+\infty}
\exp\!\left[-\tfrac{(h - c_1)^2}{2c_2}\ln\!\left(\tfrac{L}{\ell}\right)\right]
\mathrm{d}h
}.
\tag{2}
\end{equation}

To simplify notation, we define

\begin{equation}
\alpha = -\ln\!\left(\frac{\ell}{L}\right) > 0.
\tag{3}
\end{equation}

Then the expression becomes
\begin{equation}
\mathcal{P}_h^{(\ell)}(h)
=
\frac{
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right]
}{
\displaystyle \int_{-\infty}^{+\infty}
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right]
\mathrm{d}h
}.
\tag{4}
\end{equation}

The denominator is a Gaussian integral:
\begin{equation}
\int_{-\infty}^{+\infty}
e^{-\frac{\alpha}{2c_2}(h - c_1)^2}\mathrm{d}h
= \sqrt{\frac{2\pi c_2}{\alpha}}.
\tag{5}
\end{equation}

Hence,
\begin{equation}
\boxed{
\mathcal{P}_h^{(\ell)}(h)
=
\sqrt{\frac{\alpha}{2\pi c_2}}\,
\exp\!\left[-\frac{\alpha}{2c_2}(h - c_1)^2\right].
}
\tag{6}
\end{equation}

\section{Computation of the PDF using Laplace's method}

\subsection{General form of the PDF}

We now consider the probability density function of velocity increments:
\begin{equation}
\mathcal{P}_{\delta_\ell u}(x)
= \frac{1}{2\pi\sigma}\,
\sqrt{\frac{\alpha}{c_2}}
\int_{-\infty}^{+\infty}
\exp[-f(h,\alpha,x)]\,\mathrm{d}h.
\tag{7}
\end{equation}

with
\begin{equation}
f(h,\alpha,x)
= \alpha\!\left(\frac{(h-c_1)^2}{2c_2}-h\right)
+ \frac{1}{2}\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{8}
\end{equation}

\paragraph{Laplace approximation}
Let $I=[a,b]$ be an interval, $\varphi\in C^2(I,\mathbb{R})$
and $g\in C^0(I,\mathbb{C})$. Assume that:
\begin{enumerate}
    \item[(i)] For all $t>0$, $\displaystyle \int_a^b e^{t\varphi(x)} |g(x)|\,dx < +\infty$;
    \item[(ii)] $\varphi'$ vanishes at a unique point $x_0\in I$ and $\varphi''(x_0)<0$;
    \item[(iii)] $g(x_0)\neq 0$.
\end{enumerate}
Then
\[
\int_a^b e^{\,t\varphi(x)}\, g(x)\,dx
\sim
\sqrt{\frac{2\pi}{t\,|\varphi''(x_0)|}}\,
e^{\,t\varphi(x_0)}\, g(x_0)
\qquad\text{as } t\to +\infty .\]



$\\[0.8em]$
Knowing that
\begin{equation}
\forall h \in \mathbf{R},
\quad
\frac{\partial f}{\partial h}(h,\alpha,x)
= \alpha\!\left(\frac{h-c_1}{c_2}-1\right)
+ \alpha\!\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{9}
\end{equation}

we have :
\begin{equation}
\forall h \in \mathbf{R},
\quad
\frac{\partial^2 f}{\partial h^2}(h,\alpha,x)
=
\frac{\alpha}{c_2}
+ 2\alpha^2
\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha h}
\quad > 0
\tag{10}
\end{equation}

We just need to verify that $h^\star$ is such that
$\frac{\partial f}{\partial h}(h^\star,\alpha,x) = 0$
to be able to use Laplace approximation.

$\\[0.3em]$
In this case, by applying the approximation to Eq (7), we get :

\begin{equation}
\boxed{
\mathcal{P}_{\delta_\ell u}(x)
\approx
\sqrt{\frac{\alpha}{
2\pi c_2 \sigma^2 \frac{\partial^2 f}{\partial h^2}(h,\alpha,x))
}}\,
\exp[-f(h^\star,\alpha,x)].
}
\tag{11}
\end{equation}


$\\[0.1em]$
i.e
\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell u}(x)
\approx
\frac{1}
{
\sqrt{2\pi}
\sqrt{
2\alpha c_2 x^2 e^{2\alpha h^{\star}}
+ \sigma^2}}
\times
\exp\!\left[
-\alpha\!\left(
\frac{(h^\star-c_1)^2}{2c_2}-h^\star
\right)
-\frac{1}{2}
\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha h^\star}
\right]
\end{aligned}
}
\tag{12}
\end{equation}


\subsection{Case A: assuming \(h^\star = c_1\)}

As a first approximation, we can consider that :

\begin{equation}
\frac{\partial f}{\partial h}(h^\star,\alpha,x)\Big|_{h^\star=c_1}
= 0
\tag{13}
\end{equation}

$\\[0.1em]$
We can use the Laplace approximation given by Eq(12), which becomes :

\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell u}(x)
\approx
\frac{1}{\sqrt{2\pi}}
\frac{1}
{
\sqrt{
2\alpha c_2 x^2 e^{2\alpha c_{1}}
+ \sigma^{2}}}
\times
\exp\!\left[
\alpha c_1
-\frac{1}{2}\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha c_1}
\right]
\end{aligned}
}
\tag{14}
\end{equation}

\subsection{Case B: solving \(\frac{\partial f}{\partial h}(h,\alpha,x)=0\) via the Lambert \(W\) function}

From \((8)\),
\begin{equation}
\frac{\partial f}{\partial h}(h,\alpha,x)
= \alpha\!\left(\frac{h-c_1}{c_2}-1\right)
+ \alpha\!\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{15}
\end{equation}
Set to zero:
\begin{equation}
h - (c_1 + c_2)
= -c_2
\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha h}
\tag{16}
\end{equation}
Multiply by \(-2\alpha\) and let \(u=-2\alpha(h-c_1-c_2)\):
\begin{equation}
u
= 2\alpha c_2
\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)-u}
\tag{17}
\end{equation}
Consequently
\begin{equation}
u e^{u}
= 2\alpha c_2
\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}
\tag{18}
\end{equation}
Hence
\begin{equation}
\boxed{
h^\star
= (c_1 + c_2)
- \frac{1}{2\alpha}
W\!\left(
2\alpha c_2
\left(\frac{x}{\sigma}\right)^{2}
e^{2\alpha(c_1 + c_2)}
\right)
}
\tag{19}
\end{equation}
where \(W(x)\) is the Lambert \(W\) function (the inverse of \(w\mapsto we^w\)).

$\\[0.1em]$
We can now plug the obtained $h^\star$ into Eq(12).


\section{Normalized PDF}

According to Eq 12 of Chevillard et al., the $p$-th order moment of the velocity increment
is expressed as
\begin{equation}
\langle |\delta_\ell u|^p \rangle \underset{\ell \to 0}{\approx} \sigma^2 \langle |\delta|^p \rangle \left(\frac{\ell}{L}\right)^{\zeta_p},
\quad\text{with}\quad
\zeta_p = \min_h \bigl[ p h + 1 - \mathcal{D}^E(h) \bigr] \quad\text{and}\quad \delta \sim \mathcal{N}(0,1).
\tag{20}
\end{equation}

Using the Kolmogorov–Oboukhov log-normal spectrum
\begin{equation}
\mathcal{D}^E(h) = 1 - \frac{(h - c_1)^2}{2c_2},
\tag{21}
\end{equation}

the minimization problem is
\begin{equation}
\frac{\mathrm{d}}{\mathrm{d}h}
\bigl[p h + 1 - \mathcal{D}^E(h)\bigr] = 0
\quad i.e. \quad
\frac{\mathrm{d}}{\mathrm{d}h}
\bigl[p h + \frac{(h-c_1)^2}{2c_2}\bigr] = 0
\tag{22}
\end{equation}

Hence, the minimizing exponent is
\begin{equation}
h_p = c_1 - p c_2,
\tag{23}
\end{equation}
and inserting this value into Eq.~(20) gives
\begin{equation}
\zeta_p = p c_1 - \frac{p^2}{2} c_2
\tag{24}
\end{equation}

In particular, for the second-order :
\begin{equation}
\zeta_2 = 2(c_1 - c_2)
\tag{25}
\end{equation}

Knowing that $\langle |\delta|^2 \rangle = 1$, we get :
\begin{equation}
\boxed{
\langle (\delta_\ell u)^2 \rangle
\underset{\ell \to 0}{\approx}
\sigma^2 \left( \frac{\ell}{L} \right)^{2(c_1 - c_2)}.
}
\tag{26}
\end{equation}

We now define a normalized increment
\begin{equation}
\boxed{
\delta_\ell \nu
= \frac{\delta_\ell u}{\sqrt{\langle (\delta_\ell u)^2 \rangle}}
= \frac{\delta_\ell u}{\sigma \left( \frac{\ell}{L} \right)^{c_1 - c_2}}
}
\tag{27}
\end{equation}

$\\[0.3em]$
This normalization ensures that
\begin{equation}
\langle (\delta_\ell \nu)^2 \rangle = 1,
\tag{28}
\end{equation}
for all scales $\ell/L$, so that all normalized PDFs can be directly compared across scales.

$\\[0.1em]$
We thus have $\delta_\ell \nu\ = \delta_\ell u / \sigma \left( \frac{\ell}{L} \right)^{c_1 - c_2} = g(\delta_\ell u)$ where $g(x) = x / \sigma \left( \frac{\ell}{L} \right)^{c_1 - c_2}$ is monoton and differentiable.


$\\[0.1em]$
Knowing that : $\mathcal{P}_{\delta_\ell \nu}(x) = \left| \frac{1}{g'(g^{-1}(x))} \right| \mathcal{P}_{\delta_\ell u}(g^{-1}(x))$
, we have : $\mathcal{P}_{\delta_\ell \nu}(x) = \sigma \left( \frac{\ell}{L} \right)^{c_1 - c_2} \mathcal{P}_{\delta_\ell u}(\sigma \left( \frac{\ell}{L} \right)^{c_1 - c_2}x)$
i.e.

\begin{equation}
\boxed{
\mathcal{P}_{\delta_\ell \nu}(x) = \sigma e^{\alpha (c_2 - c_1)} \mathcal{P}_{\delta_\ell u}(\sigma e^{\alpha (c_2 - c_1)}x)
}
\tag{29}
\end{equation}

$\\[0.1em]$
We then obtain from Eq (12) the normalized form of the PDF :

\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell \nu}(x)
\approx
\frac{e^{\alpha(c_2 - c_1)}}{
\sqrt{2\pi}
\sqrt{
2 \alpha c_2 x^2 e^{2\alpha (h^\star + c_2 - c_1)} + 1
}
}
\times
\exp\!\left[
-\alpha\!\left(
\frac{(h^\star-c_1)^2}{2c_2}-h^\star
\right)
- \frac{1}{2}
x^{2}
e^{2\alpha (h^\star + c_2 - c_1)}
\right]
\end{aligned}
}
\tag{30}
\end{equation}

which is valid if $h^\star$ is such that $\frac{\partial f}{\partial h}(h^\star,\alpha,\sigma e^{\alpha (c_2-c_1)} x)=0$.


\subsection{Case A: assuming \(h^\star = c_1\)}

If we consider that
$\frac{\partial f}{\partial h}(h^\star,\alpha,\sigma e^{\alpha (c_2-c_1)} x)\Big|_{h^\star=c_1} = 0$,
then we get from Eq (30) :

\begin{equation}
\boxed{
\begin{aligned}
\mathcal{P}_{\delta_\ell \nu}(x)
\approx
\frac{1}{
\sqrt{2\pi}
\sqrt{
2 \alpha c_2\,
x^2
e^{2\alpha c_2}
+ 1
}
}
\times
\exp\!\left[
\alpha c_2
-\frac{1}{2}
x^2
e^{2\alpha c_2}
\right]
\end{aligned}
}
\tag{31}
\end{equation}



\subsection{Case B: solving \(\frac{\partial f}{\partial h}(h,\alpha,\sigma e^{\alpha (c_2-c_1)} x)=0\) via the Lambert \(W\) function}

We want to find \(h^\star\) such that $\frac{\partial f}{\partial h}(h^\star,\alpha,\sigma e^{\alpha (c_2-c_1)}x) = 0$.
In Eq(19), by considering that \(h^\star\) is a function of x, we just need to replace x by \( \sigma e^{\alpha (c_2-c_1)} x\). We get :

\begin{equation}
\boxed{
h^\star
= (c_1 + c_2)
- \frac{1}{2\alpha}
W\!\left(
2\alpha c_2 x^2 e^{4\alpha c_2}
\right)
}
\tag{32}
\end{equation}

We can now plug the obtained $h^\star$ into Eq(30). Let $W=W\!\left( 2\alpha c_2 x^2 e^{4\alpha c_2} \right)$. We get :

\begin{equation}
\boxed{
\mathcal{P}_{\delta_\ell \nu}(x)
\approx
\frac{1}{
\sqrt{2\pi}\sqrt{2 \alpha c_2 x^2 e^{4 \alpha c_2-W} + 1}}
\times
\exp\left[
-\frac{\alpha}{2} \left( \frac{1}{c_2} \left( \frac{W}{2\alpha} \right)^2 - 3c_2 \right)
-\frac{1}{2} x^2 e^{4 \alpha c_2 - W}
\right]
}
\tag{33}
\end{equation}

i.e.

\begin{equation}
\boxed{
\mathcal{P}_{\delta_\ell \nu}(x)
\approx
\frac{e^\frac{3}{2}c_2 \alpha}{\sqrt{2\pi}}
\frac{e^{-\frac{2W + W^2}{8c_2 \alpha}}}{\sqrt{1+W}}
}
\tag{34}
\end{equation}




\section{Entropy of the normalized PDF}

We now compute the Shannon entropy of the normalized increment PDF
\(\mathcal{P}_{\delta_\ell \nu}\) obtained in Eq. (28). By definition,
\begin{equation}
S(\delta_\ell \nu)
=
- \int_{-\infty}^{+\infty}
\mathcal{P}_{\delta_\ell \nu}(x)\,
\ln\!\bigl[\mathcal{P}_{\delta_\ell \nu}(x)\bigr]\,
\mathrm{d}x
\tag{35}
\end{equation}
